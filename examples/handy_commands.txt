SELECT *
FROM read_parquet(
    's3://cdc-events/**/*.parquet',
    hive_partitioning = true
)
order by __source_ts_ms desc;


SELECT __op, COUNT(*)
FROM read_parquet('s3://cdc-events/**/*.parquet', hive_partitioning = true)
WHERE year = '2026' AND month = '06'
GROUP BY __op;

WITH ranked AS (
    SELECT *,
        ROW_NUMBER() OVER (
            PARTITION BY id
            ORDER BY __source_ts_ms DESC, updated_at DESC
        ) AS rn
    FROM read_parquet('s3://cdc-events/**/*.parquet', hive_partitioning = true)
)
SELECT * EXCLUDE (rn)
FROM ranked
WHERE rn = 1
  AND __op <> 'd';
